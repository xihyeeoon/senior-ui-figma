function korean(n){
  if(!n)return '영 원';
  const digits=['','일','이','삼','사','오','육','칠','팔','구'];
  function part(v){
    const units=['','십','백','천'];
    let t='';
    for(let i=3;i>=0;i--){
      const d=Math.floor(v/Math.pow(10,i))%10;
      if(d)t+=(d===1&&i>0?'':digits[d])+units[i];
    }
    return t;
  }
  const high=Math.floor(n/10000),low=n%10000;
  return (high?(high===1?'일':part(high))+'만 ':'')+part(low)+' 원';
}
